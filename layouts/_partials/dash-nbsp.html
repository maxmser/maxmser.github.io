{{- /* Expects text or HTML. Puts a no-break space before every spaced en or em dash, so a line never begins with a dash (Bringhurst 5.2.1; rule 1.3 of the CV typography rules). Covers the character itself and Hugo's –/—. Also binds a spaced multiplication sign between numbers ("1179 × 2556"), so the dimensions never break apart. */ -}} {{- $s := string . }} {{- range slice "–" "—" "–" "—" }} {{- $s = replace $s (printf " %s " .) (printf " %s " .) }} {{- end }} {{- $s = replaceRE `(\d) (×|×) (\d)` "${1}\u00a0${2}\u00a0${3}" $s }} {{- return $s }}
