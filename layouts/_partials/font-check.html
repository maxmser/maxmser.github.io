{{- /* Stops the build if a page uses a character that the web font does not contain. Without this check such a character would silently appear in another font. The allowed characters come from data/font.toml, the same list that tools/build-fonts.py uses to build the font. To allow a character, add it there and rebuild the font. */ -}} {{- $class := "" }} {{- range hugo.Data.font.unicodes }} {{- $parts := split . "-" }} {{- $class = printf `%s\x{%s}` $class (index $parts 0) }} {{- if gt (len $parts) 1 }}{{ $class = printf `%s-\x{%s}` $class (index $parts 1) }}{{ end }} {{- end }} {{- $texts := slice site.Title .Title (.Params.summary | default "") (.Params.status | default "") (delimit (.Params.stack | default slice) " ") (.Content | plainify | htmlUnescape) }} {{- range .Params.more_projects }}{{ $texts = $texts | append .name }}{{ end }} {{- with findRE (printf `[^%s\s]` $class) (delimit $texts " ") | uniq }} {{- errorf "%s: characters not in the web font: %s (add them to data/font.toml and run tools/build-fonts.py)" $.Path (delimit . " ") }} {{- end }}
